cmake_minimum_required(VERSION 3.24)
project(FileManagerPreviewAdmission LANGUAGES CXX)
find_package(GUIForms CONFIG REQUIRED COMPONENTS Core)
add_executable(preview_admission probe.cpp)
target_compile_features(preview_admission PRIVATE cxx_std_20)
set_target_properties(preview_admission PROPERTIES CXX_EXTENSIONS OFF)
target_link_libraries(preview_admission PRIVATE GUIForms::Core)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(preview_admission PRIVATE -Wall -Wextra -Werror -Wpedantic)
endif()
